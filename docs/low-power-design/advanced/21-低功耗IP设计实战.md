# 第21章 低功耗IP设计实战

## 21.1 引言

本章通过一个完整的低功耗子系统设计实例——**PMU控制器 (Power Management Unit Controller)**，串联前面所有章节的知识点，展示从架构定义、RTL编码、UPF编写、仿真验证到综合实现的全流程。这是一个"从0到1"的实战项目，旨在帮助读者将理论转化为工程实践能力。

## 21.2 项目概述

### 21.2.1 设计目标

```
PMU控制器设计目标:

功能:
  管理一个包含3个电源域的小型SoC的电源状态

电源域:
  PD_AON:   常开域 (PMU自身+唤醒逻辑)
  PD_CORE:  可关断域 (处理器核心)
  PD_PERIPH: 可关断域 (外设模块)

支持的系统状态:
  S0_ACTIVE:  所有域开启
  S1_IDLE:    CORE降频, PERIPH可选关断
  S2_STANDBY: CORE保持(retention), PERIPH关断
  S3_DEEPSLEEP: CORE关断, PERIPH关断

接口:
  - APB从接口 (软件配置)
  - 电源控制信号输出 (power switch, isolation, retention)
  - 唤醒中断输入
  - 时钟控制信号
  - PMIC/Regulator接口
```

module pmu_ctrl_top (

### 21.3.2 电源状态机 (PSM)

```verilog
///////////////////////////////////////////////
// Power State Machine
///////////////////////////////////////////////
module power_state_machine (
    input  wire        clk,
    input  wire        rst_n,
    
    // Control inputs
    input  wire [1:0]  target_state,  // From register file
    input  wire        state_change_req,
    input  wire        wakeup_event,
    
    // Sequence controller interface
    output reg         seq_start,
    output reg  [1:0]  seq_target,
    input  wire        seq_done,
    
    // Status
    output reg  [1:0]  current_state,
    output wire        state_transitioning
);

    // State encoding
    localparam S0_ACTIVE    = 2'b00;
    localparam S1_IDLE      = 2'b01;
    localparam S2_STANDBY   = 2'b10;
    localparam S3_DEEPSLEEP = 2'b11;
    
    // FSM states
    localparam PSM_IDLE       = 3'b000;
    localparam PSM_PWR_DOWN   = 3'b001;
    localparam PSM_WAIT_DOWN  = 3'b010;
    localparam PSM_SLEEPING   = 3'b011;
    localparam PSM_PWR_UP     = 3'b100;
    localparam PSM_WAIT_UP    = 3'b101;
    
    reg [2:0] psm_state, psm_next;
    
    assign state_transitioning = (psm_state != PSM_IDLE) && 
                                  (psm_state != PSM_SLEEPING);
    
    // State register
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            psm_state <= PSM_IDLE;
        else
            psm_state <= psm_next;
    end
    
    // Next state logic
    always @(*) begin
        psm_next = psm_state;
        seq_start = 1'b0;
        seq_target = current_state;
        
        case (psm_state)
            PSM_IDLE: begin
                if (state_change_req && 
                    target_state > current_state) begin
                    // Going to lower power state
                    psm_next = PSM_PWR_DOWN;
                end
                else if (state_change_req && 
                         target_state < current_state) begin
                    // Going to higher power state
                    psm_next = PSM_PWR_UP;
                end
            end
            
            PSM_PWR_DOWN: begin
                seq_start = 1'b1;
                seq_target = target_state;
                psm_next = PSM_WAIT_DOWN;
            end
            
            PSM_WAIT_DOWN: begin
                if (seq_done) begin
                    psm_next = PSM_SLEEPING;
                end
            end
            
            PSM_SLEEPING: begin
                // Wait for wakeup event or software request
                if (wakeup_event || 
                    (state_change_req && 
                     target_state < current_state)) begin
                    psm_next = PSM_PWR_UP;
                end
            end
            
            PSM_PWR_UP: begin
                seq_start = 1'b1;
                seq_target = (wakeup_event) ? S0_ACTIVE : target_state;
                psm_next = PSM_WAIT_UP;
            end
            
            PSM_WAIT_UP: begin
                if (seq_done) begin
                    psm_next = PSM_IDLE;
                end
            end
            
            default: psm_next = PSM_IDLE;
        endcase
    end
    
    // Current state update
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            current_state <= S0_ACTIVE;
        else if (psm_state == PSM_WAIT_DOWN && seq_done)
            current_state <= target_state;
        else if (psm_state == PSM_WAIT_UP && seq_done)
            current_state <= seq_target;
    end

endmodule
```

### 21.3.3 序列控制器

```verilog
///////////////////////////////////////////////
// Power Sequence Controller
// 控制关断和恢复的正确时序
///////////////////////////////////////////////
module sequence_controller (
    input  wire        clk,
    input  wire        rst_n,
    
    // Control
    input  wire        seq_start,
    input  wire [1:0]  seq_target,
    input  wire [1:0]  current_state,
    output reg         seq_done,
    
    // Power control outputs
    output reg         core_pwr_en,
    output reg         core_iso_en,
    output reg         core_ret_save,
    output reg         core_ret_restore,
    output reg         core_clk_en,
    output reg         periph_pwr_en,
    output reg         periph_iso_en,
    output reg         periph_clk_en,
    
    // Acknowledges
    input  wire        core_pwr_ack,
    input  wire        periph_pwr_ack,
    
    // PMIC
    output reg         pmic_req,
    output reg  [3:0]  pmic_voltage,
    input  wire        pmic_ack
);

    // Sequence states
    localparam SEQ_IDLE          = 4'd0;
    // Power-down sequence
    localparam SEQ_CLK_OFF       = 4'd1;
    localparam SEQ_ISO_ON        = 4'd2;
    localparam SEQ_RET_SAVE      = 4'd3;
    localparam SEQ_RET_SAVE_DONE = 4'd4;
    localparam SEQ_PWR_OFF       = 4'd5;
    localparam SEQ_PWR_OFF_WAIT  = 4'd6;
    // Power-up sequence
    localparam SEQ_PWR_ON        = 4'd7;
    localparam SEQ_PWR_ON_WAIT   = 4'd8;
    localparam SEQ_RET_RESTORE   = 4'd9;
    localparam SEQ_RET_REST_DONE = 4'd10;
    localparam SEQ_ISO_OFF       = 4'd11;
    localparam SEQ_CLK_ON        = 4'd12;
    localparam SEQ_DONE          = 4'd13;
    
    reg [3:0] seq_state, seq_next;
    reg       going_down; // 1=power down, 0=power up
    
    // Delay counter for timing requirements
    reg [7:0] delay_cnt;
    wire      delay_done = (delay_cnt == 8'd0);
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            delay_cnt <= 8'd0;
        else if (seq_state != seq_next)
            delay_cnt <= 8'd10; // 10 cycle delay between steps
        else if (delay_cnt != 8'd0)
            delay_cnt <= delay_cnt - 8'd1;
    end
    
    // State register
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            seq_state <= SEQ_IDLE;
        else
            seq_state <= seq_next;
    end
    
    // Direction register
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            going_down <= 1'b0;
        else if (seq_state == SEQ_IDLE && seq_start)
            going_down <= (seq_target > current_state);
    end
    
    // Next state logic
    always @(*) begin
        seq_next = seq_state;
        seq_done = 1'b0;
        
        case (seq_state)
            SEQ_IDLE: begin
                if (seq_start) begin
                    if (seq_target > current_state)
                        seq_next = SEQ_CLK_OFF;   // Power down
                    else
                        seq_next = SEQ_PWR_ON;     // Power up
                end
            end
            
            // === Power-Down Sequence ===
            // Step 1: 停止时钟
            SEQ_CLK_OFF: begin
                if (delay_done) seq_next = SEQ_ISO_ON;
            end
            
            // Step 2: 使能隔离
            SEQ_ISO_ON: begin
                if (delay_done) seq_next = SEQ_RET_SAVE;
            end
            
            // Step 3: 保存状态 (retention save pulse)
            SEQ_RET_SAVE: begin
                seq_next = SEQ_RET_SAVE_DONE;
            end
            
            SEQ_RET_SAVE_DONE: begin
                if (delay_done) seq_next = SEQ_PWR_OFF;
            end
            
            // Step 4: 关断电源
            SEQ_PWR_OFF: begin
                seq_next = SEQ_PWR_OFF_WAIT;
            end
            
            SEQ_PWR_OFF_WAIT: begin
                // Wait for power switch acknowledge
                if (!core_pwr_ack && !periph_pwr_ack)
                    seq_next = SEQ_DONE;
            end
            
            // === Power-Up Sequence ===
            // Step 1: 打开电源
            SEQ_PWR_ON: begin
                seq_next = SEQ_PWR_ON_WAIT;
            end
            
            SEQ_PWR_ON_WAIT: begin
                // Wait for power stable
                if (core_pwr_ack || periph_pwr_ack)
                    if (delay_done) seq_next = SEQ_RET_RESTORE;
            end
            
            // Step 2: 恢复状态
            SEQ_RET_RESTORE: begin
                seq_next = SEQ_RET_REST_DONE;
            end
            
            SEQ_RET_REST_DONE: begin
                if (delay_done) seq_next = SEQ_ISO_OFF;
            end
            
            // Step 3: 关闭隔离
            SEQ_ISO_OFF: begin
                if (delay_done) seq_next = SEQ_CLK_ON;
            end
            
            // Step 4: 恢复时钟
            SEQ_CLK_ON: begin
                if (delay_done) seq_next = SEQ_DONE;
            end
            
            SEQ_DONE: begin
                seq_done = 1'b1;
                seq_next = SEQ_IDLE;
            end
            
            default: seq_next = SEQ_IDLE;
        endcase
    end
    
    // Output logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            core_pwr_en      <= 1'b1;  // Power on at reset
            core_iso_en      <= 1'b0;  // No isolation
            core_ret_save    <= 1'b0;
            core_ret_restore <= 1'b0;
            core_clk_en      <= 1'b1;  // Clock enabled
            periph_pwr_en    <= 1'b1;
            periph_iso_en    <= 1'b0;
            periph_clk_en    <= 1'b1;
            pmic_req         <= 1'b0;
            pmic_voltage     <= 4'd8;  // Nominal voltage
        end
        else begin
            // Default: deassert pulses
            core_ret_save    <= 1'b0;
            core_ret_restore <= 1'b0;
            
            case (seq_state)
                SEQ_CLK_OFF: begin
                    core_clk_en  <= 1'b0;
                    periph_clk_en <= 1'b0;
                end
                
                SEQ_ISO_ON: begin
                    core_iso_en   <= 1'b1;
                    periph_iso_en <= 1'b1;
                end
                
                SEQ_RET_SAVE: begin
                    core_ret_save <= 1'b1; // Pulse
                end
                
                SEQ_PWR_OFF: begin
                    core_pwr_en   <= 1'b0;
                    periph_pwr_en <= 1'b0;
                end
                
                SEQ_PWR_ON: begin
                    core_pwr_en   <= 1'b1;
                    periph_pwr_en <= 1'b1;
                end
                
                SEQ_RET_RESTORE: begin
                    core_ret_restore <= 1'b1; // Pulse
                end
                
                SEQ_ISO_OFF: begin
                    core_iso_en   <= 1'b0;
                    periph_iso_en <= 1'b0;
                end
                
                SEQ_CLK_ON: begin
                    core_clk_en   <= 1'b1;
                    periph_clk_en <= 1'b1;
                end
            endcase
        end
    end

endmodule
```

### 21.3.4 唤醒控制器

```verilog
///////////////////////////////////////////////
// Wakeup Controller
///////////////////////////////////////////////
module wakeup_controller (
    input  wire        clk,
    input  wire        rst_n,
    
    // Wakeup sources
    input  wire [7:0]  wakeup_src,
    input  wire        rtc_alarm,
    
    // Wakeup mask (from register file)
    input  wire [7:0]  wakeup_mask,
    input  wire        rtc_mask,
    
    // Output
    output wire        wakeup_event,
    output wire [8:0]  wakeup_status
);

    // Edge detection for wakeup sources
    reg [7:0] wakeup_src_d;
    wire [7:0] wakeup_edges;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            wakeup_src_d <= 8'd0;
        else
            wakeup_src_d <= wakeup_src;
    end
    
    assign wakeup_edges = wakeup_src & ~wakeup_src_d;
    
    // Apply mask
    wire [7:0] masked_wakeup = wakeup_edges & wakeup_mask;
    wire       masked_rtc    = rtc_alarm & rtc_mask;
    
    // Combined wakeup event
    assign wakeup_event = |masked_wakeup | masked_rtc;
    
    // Wakeup status (for software to read)
    assign wakeup_status = {masked_rtc, masked_wakeup};

endmodule
```

    -domain PD_PERIPH \
    -input_supply_port  {vin  VDD_PRPH} \

    -domain PD_CORE \

    -clamp_value 0 \

    -domain PD_CORE \

## 21.5 验证

### 21.5.1 PA-Sim Testbench 关键场景

```systemverilog
///////////////////////////////////////////////
// PMU Controller Testbench
///////////////////////////////////////////////
module tb_pmu_ctrl;

    // ... (信号定义和DUT实例化省略) ...

    // 测试场景1: 正常关断-恢复序列
    task test_power_cycle();
        $display("=== Test: Power Cycle ===");
        
        // 1. 写入目标状态 = S2_STANDBY
        apb_write(CTRL_REG, 32'h0000_0002);
        
        // 2. 触发状态切换
        apb_write(CTRL_REG, 32'h8000_0002); // bit31=start
        
        // 3. 等待切换完成
        wait(dut.seq_done);
        
        // 4. 检查信号状态
        assert(dut.core_pwr_en == 0)
            else $error("Core power should be off!");
        assert(dut.core_iso_en == 1)
            else $error("Core isolation should be on!");
        assert(dut.periph_pwr_en == 0)
            else $error("Periph power should be off!");
        
        // 5. 模拟唤醒
        #1000;
        wakeup_src[0] = 1'b1;
        
        // 6. 等待恢复
        wait(dut.current_state == 2'b00);
        
        // 7. 验证恢复后信号
        assert(dut.core_pwr_en == 1)
            else $error("Core power should be on!");
        assert(dut.core_iso_en == 0)
            else $error("Core isolation should be off!");
            
        $display("=== Power Cycle Test PASSED ===");
    endtask

    // 测试场景2: 隔离时序检查
    task test_isolation_timing();
        $display("=== Test: Isolation Timing ===");
        
        // 关断时: ISO enable 必须在 PWR off 之前
        fork
            begin
                @(posedge dut.core_iso_en);
                iso_time = $time;
            end
            begin
                @(negedge dut.core_pwr_en);
                pwr_off_time = $time;
            end
        join
        
        assert(iso_time < pwr_off_time)
            else $error("Isolation must be enabled before power off!");
        
        $display("ISO→PWR_OFF timing: %0t ns", 
                 pwr_off_time - iso_time);
    endtask

    // 测试场景3: Retention 数据完整性
    task test_retention_integrity();
        $display("=== Test: Retention Integrity ===");
        
        // 记录关断前的寄存器值
        saved_data = dut.u_cpu_core.gpr[1]; // 保存通用寄存器
        
        // 执行关断-恢复
        trigger_standby();
        wait_for_sleep();
        trigger_wakeup();
        wait_for_active();
        
        // 比较恢复后的值
        assert(dut.u_cpu_core.gpr[1] == saved_data)
            else $error("Retention data mismatch! Expected %h, Got %h",
                       saved_data, dut.u_cpu_core.gpr[1]);
                       
        $display("=== Retention Test PASSED ===");
    endtask

endmodule
```

    -positive_edge_logic {integrated} \
    -minimum_bitwidth 4

# 综合

check_mv_design -verbose

# 报告
