# 第22章 功耗分析脚本开发

## 22.1 引言

在实际低功耗设计项目中，功耗分析涉及大量的数据处理、报告解析和趋势追踪工作。手动操作不仅效率低下，而且容易出错。本章介绍如何使用 **Python 和 Tcl** 开发功耗分析自动化脚本，覆盖功耗报告解析、趋势分析、回归对比、可视化等实用场景。

## 22.2 功耗报告解析

### 22.2.1 PrimeTime PX 报告解析器

```python
#!/usr/bin/env python3
"""
PrimeTime PX Power Report Parser
解析PTPX生成的功耗报告，提取关键数据
"""

import re
import csv
from pathlib import Path
from dataclasses import dataclass, field
from typing import List, Dict, Optional


@dataclass
class ModulePower:
    """单个模块的功耗数据"""
    hierarchy: str
    internal_power: float  # mW
    switching_power: float  # mW
    leakage_power: float  # mW
    total_power: float  # mW
    percentage: float  # %


@dataclass
class PowerReport:
    """完整功耗报告"""
    design_name: str = ""
    voltage: float = 0.0
    temperature: float = 0.0
    total_power: float = 0.0
    modules: List[ModulePower] = field(default_factory=list)


def parse_ptpx_report(filepath: str) -> PowerReport:
    """解析PTPX功耗报告文件"""
    report = PowerReport()
    
    with open(filepath, 'r') as f:
        content = f.read()
    
    # 提取设计名称
    match = re.search(r'Design\s*:\s*(\S+)', content)
    if match:
        report.design_name = match.group(1)
    
    # 提取电压
    match = re.search(r'Voltage\s*:\s*([\d.]+)\s*V', content)
    if match:
        report.voltage = float(match.group(1))
    
    # 提取温度
    match = re.search(r'Temperature\s*:\s*([\d.]+)\s*C', content)
    if match:
        report.temperature = float(match.group(1))
    
    # 解析层次化功耗表格
    # 典型格式:
    # Hierarchy  Internal  Switching  Leakage  Total    %
    # u_cpu      10.5      8.3        2.1      20.9     45.2%
    table_pattern = re.compile(
        r'(\S+)\s+'           # hierarchy
        r'([\d.]+(?:e[+-]?\d+)?)\s+'  # internal
        r'([\d.]+(?:e[+-]?\d+)?)\s+'  # switching
        r'([\d.]+(?:e[+-]?\d+)?)\s+'  # leakage
        r'([\d.]+(?:e[+-]?\d+)?)\s+'  # total
        r'([\d.]+)%'          # percentage
    )
    
    for match in table_pattern.finditer(content):
        module = ModulePower(
            hierarchy=match.group(1),
            internal_power=float(match.group(2)),
            switching_power=float(match.group(3)),
            leakage_power=float(match.group(4)),
            total_power=float(match.group(5)),
            percentage=float(match.group(6))
        )
        report.modules.append(module)
    
    # 总功耗
    if report.modules:
        report.total_power = sum(m.total_power for m in report.modules)
    
    return report


def export_to_csv(report: PowerReport, output_path: str):
    """导出功耗数据到CSV"""
    with open(output_path, 'w', newline='') as f:
        writer = csv.writer(f)
        writer.writerow([
            'Hierarchy', 'Internal(mW)', 'Switching(mW)',
            'Leakage(mW)', 'Total(mW)', 'Percentage(%)'
        ])
        for m in report.modules:
            writer.writerow([
                m.hierarchy, f'{m.internal_power:.4f}',
                f'{m.switching_power:.4f}', f'{m.leakage_power:.4f}',
                f'{m.total_power:.4f}', f'{m.percentage:.1f}'
            ])


# 使用示例
if __name__ == '__main__':
    report = parse_ptpx_report('power_report.rpt')
    print(f"Design: {report.design_name}")
    print(f"Total Power: {report.total_power:.2f} mW")
    print(f"\nTop 5 power consumers:")
    sorted_modules = sorted(report.modules,
                           key=lambda m: m.total_power,
                           reverse=True)
    for m in sorted_modules[:5]:
        print(f"  {m.hierarchy:30s} {m.total_power:8.3f} mW "
              f"({m.percentage:.1f}%)")
    export_to_csv(report, 'power_data.csv')
```

## 22.3 功耗趋势分析

### 22.3.1 多版本功耗趋势追踪

```python
#!/usr/bin/env python3
"""
Power Trend Tracker
追踪多个设计版本之间的功耗变化趋势
"""

import json
from datetime import datetime
from pathlib import Path
from typing import Dict, List


class PowerTrendTracker:
    """功耗趋势追踪器"""
    
    def __init__(self, db_path: str = 'power_trend.json'):
        self.db_path = Path(db_path)
        self.data = self._load_db()
    
    def _load_db(self) -> Dict:
        """加载历史数据"""
        if self.db_path.exists():
            with open(self.db_path) as f:
                return json.load(f)
        return {'versions': []}
    
    def _save_db(self):
        """保存数据"""
        with open(self.db_path, 'w') as f:
            json.dump(self.data, f, indent=2)
    
    def add_version(self, version: str, report_path: str,
                    commit_hash: str = ''):
        """添加新版本的功耗数据"""
        # 解析报告(使用前面的解析器)
        report = parse_ptpx_report(report_path)
        
        entry = {
            'version': version,
            'timestamp': datetime.now().isoformat(),
            'commit': commit_hash,
            'total_power': report.total_power,
            'breakdown': {
                'internal': sum(m.internal_power 
                              for m in report.modules),
                'switching': sum(m.switching_power 
                               for m in report.modules),
                'leakage': sum(m.leakage_power 
                             for m in report.modules),
            },
            'top_modules': [
                {
                    'name': m.hierarchy,
                    'power': m.total_power
                }
                for m in sorted(report.modules,
                               key=lambda x: x.total_power,
                               reverse=True)[:10]
            ]
        }
        
        self.data['versions'].append(entry)
        self._save_db()
        
        # 检查功耗回归
        self._check_regression(entry)
    
    def _check_regression(self, current):
        """检查功耗是否回归(恶化)"""
        versions = self.data['versions']
        if len(versions) < 2:
            return
        
        previous = versions[-2]
        delta = current['total_power'] - previous['total_power']
        pct = delta / previous['total_power'] * 100
        
        if pct > 5.0:
            print(f"⚠️  POWER REGRESSION DETECTED!")
            print(f"    Previous: {previous['total_power']:.2f} mW "
                  f"({previous['version']})")
            print(f"    Current:  {current['total_power']:.2f} mW "
                  f"({current['version']})")
            print(f"    Delta:    +{delta:.2f} mW (+{pct:.1f}%)")
        elif pct < -5.0:
            print(f"✅ Power improved by {-delta:.2f} mW ({-pct:.1f}%)")
        else:
            print(f"ℹ️  Power stable: {delta:+.2f} mW ({pct:+.1f}%)")
    
    def generate_report(self) -> str:
        """生成趋势报告"""
        lines = []
        lines.append("=" * 60)
        lines.append("  Power Trend Report")
        lines.append("=" * 60)
        
        for v in self.data['versions']:
            bd = v['breakdown']
            lines.append(
                f"\n  Version: {v['version']}"
                f"  ({v['timestamp'][:10]})"
            )
            lines.append(f"  Total: {v['total_power']:.2f} mW")
            lines.append(
                f"    Internal:  {bd['internal']:.2f} mW  "
                f"Switching: {bd['switching']:.2f} mW  "
                f"Leakage: {bd['leakage']:.2f} mW"
            )
        
        return '\n'.join(lines)
```

## 22.4 时钟门控分析脚本

### 22.4.1 CG 覆盖率分析

```python
#!/usr/bin/env python3
"""
Clock Gating Coverage Analyzer
分析时钟门控覆盖率报告，找出未门控寄存器
"""

import re
from collections import defaultdict
from typing import Dict, List, Tuple


def parse_cg_report(filepath: str) -> Dict:
    """解析DC的时钟门控报告"""
    result = {
        'total_regs': 0,
        'gated_regs': 0,
        'ungated_regs': 0,
        'coverage': 0.0,
        'modules': [],
        'ungated_list': []
    }
    
    with open(filepath, 'r') as f:
        content = f.read()
    
    # 解析模块级统计
    # Module   Gated  Ungated  Coverage
    module_pat = re.compile(
        r'(\S+)\s+(\d+)\s+(\d+)\s+([\d.]+)%'
    )
    
    for match in module_pat.finditer(content):
        module = {
            'name': match.group(1),
            'gated': int(match.group(2)),
            'ungated': int(match.group(3)),
            'coverage': float(match.group(4))
        }
        result['modules'].append(module)
        result['total_regs'] += module['gated'] + module['ungated']
        result['gated_regs'] += module['gated']
        result['ungated_regs'] += module['ungated']
    
    if result['total_regs'] > 0:
        result['coverage'] = (result['gated_regs'] / 
                              result['total_regs'] * 100)
    
    return result


def find_optimization_opportunities(cg_data: Dict,
                                     threshold: float = 85.0
                                     ) -> List[Dict]:
    """找出需要优化的模块"""
    opportunities = []
    
    for mod in cg_data['modules']:
        if mod['coverage'] < threshold and mod['ungated'] > 10:
            savings_estimate = mod['ungated'] * 0.01  # 粗估mW
            opportunities.append({
                'module': mod['name'],
                'coverage': mod['coverage'],
                'ungated_count': mod['ungated'],
                'potential_savings_mw': savings_estimate
            })
    
    return sorted(opportunities,
                 key=lambda x: x['ungated_count'],
                 reverse=True)


def generate_cg_summary(cg_data: Dict):
    """生成时钟门控分析摘要"""
    print(f"{'='*50}")
    print(f"Clock Gating Coverage Summary")
    print(f"{'='*50}")
    print(f"Total Registers:   {cg_data['total_regs']:>8d}")
    print(f"Gated Registers:   {cg_data['gated_regs']:>8d}")
    print(f"Ungated Registers: {cg_data['ungated_regs']:>8d}")
    print(f"Overall Coverage:  {cg_data['coverage']:>7.1f}%")
    
    target = 90.0
    if cg_data['coverage'] >= target:
        print(f"\n✅ Coverage meets target ({target}%)")
    else:
        gap = cg_data['ungated_regs'] - int(
            cg_data['total_regs'] * (1 - target/100))
        print(f"\n⚠️  Coverage below target ({target}%)")
        print(f"    Need to gate {gap} more registers")
    
    # 低覆盖率模块
    opportunities = find_optimization_opportunities(cg_data)
    if opportunities:
        print(f"\nModules needing optimization:")
        for opp in opportunities[:10]:
            print(f"  {opp['module']:30s} "
                  f"Coverage: {opp['coverage']:5.1f}%  "
                  f"Ungated: {opp['ungated_count']:5d}")
```

## 22.5 Multi-Vt 分析脚本

### 22.5.1 Vt 分布分析

```python
#!/usr/bin/env python3
"""
Multi-Vt Distribution Analyzer
分析多阈值电压单元的分布
"""

import re
from collections import Counter
from typing import Dict


def parse_vt_report(filepath: str) -> Dict:
    """解析Multi-Vt分布报告"""
    vt_counts = Counter()
    
    with open(filepath, 'r') as f:
        for line in f:
            # 匹配: cell_type  count  percentage
            match = re.match(
                r'\s*(u?[HL]VT|SVT)\s+(\d+)\s+([\d.]+)%', line)
            if match:
                vt_type = match.group(1)
                count = int(match.group(2))
                vt_counts[vt_type] = count
    
    total = sum(vt_counts.values())
    
    return {
        'counts': dict(vt_counts),
        'total': total,
        'percentages': {
            vt: count/total*100 if total > 0 else 0
            for vt, count in vt_counts.items()
        }
    }


def check_vt_constraints(vt_data: Dict,
                          max_lvt_pct: float = 10.0,
                          min_hvt_pct: float = 60.0):
    """检查Vt分布是否满足约束"""
    pcts = vt_data['percentages']
    
    print(f"{'='*40}")
    print(f"Multi-Vt Distribution Check")
    print(f"{'='*40}")
    
    for vt_type in ['uHVT', 'HVT', 'SVT', 'LVT', 'uLVT']:
        if vt_type in pcts:
            bar = '█' * int(pcts[vt_type] / 2)
            print(f"  {vt_type:5s}: {pcts[vt_type]:5.1f}%  "
                  f"{vt_data['counts'][vt_type]:>6d} cells  {bar}")
    
    # 检查LVT约束
    lvt_total = sum(pcts.get(vt, 0) for vt in ['LVT', 'uLVT'])
    hvt_total = sum(pcts.get(vt, 0) for vt in ['HVT', 'uHVT'])
    
    print(f"\nConstraint Checks:")
    if lvt_total <= max_lvt_pct:
        print(f"  ✅ LVT+uLVT: {lvt_total:.1f}% "
              f"(<= {max_lvt_pct}% limit)")
    else:
        print(f"  ❌ LVT+uLVT: {lvt_total:.1f}% "
              f"EXCEEDS {max_lvt_pct}% limit!")
    
    if hvt_total >= min_hvt_pct:
        print(f"  ✅ HVT+uHVT: {hvt_total:.1f}% "
              f"(>= {min_hvt_pct}% target)")
    else:
        print(f"  ⚠️  HVT+uHVT: {hvt_total:.1f}% "
              f"below {min_hvt_pct}% target")
```

```tcl
###############################################
# UPF一致性自动检查脚本
###############################################

proc check_upf_consistency {} {
    puts "=== UPF Consistency Check ==="
    set errors 0
    set warnings 0
    
    # 检查1: 所有跨域信号都有隔离
    puts "\nCheck 1: Isolation coverage..."
    set crossing_signals [get_crossing_signals -all]
    set isolated_signals [get_isolated_signals -all]
    
    foreach sig $crossing_signals {
        if {[lsearch $isolated_signals $sig] == -1} {
            puts "  ERROR: Signal $sig crosses domain without isolation!"
            incr errors
        }
    }
    
    # 检查2: 可关断域寄存器有保持策略
    puts "\nCheck 2: Retention coverage..."
    set switchable_domains [get_power_domains -filter "is_switchable == true"]
    foreach domain $switchable_domains {
        set regs [get_cells -hierarchical -filter \
            "power_domain == $domain && is_sequential == true"]
        set ret_regs [get_retention_cells -domain $domain]
        
        set total [sizeof_collection $regs]
        set retained [sizeof_collection $ret_regs]
        
        if {$retained < $total} {
            puts "  WARNING: Domain $domain has [expr {$total - $retained}] "\
                 "registers without retention"
            incr warnings
        }
    }
    
    # 检查3: 隔离/保持控制信号来自常开域
    puts "\nCheck 3: Control signal domains..."
    set iso_controls [get_isolation_controls -all]
    foreach ctrl $iso_controls {
        set ctrl_domain [get_attribute $ctrl power_domain]
        if {[get_attribute $ctrl_domain is_switchable] == "true"} {
            puts "  ERROR: Isolation control $ctrl is in switchable domain!"
            incr errors
        }
    }
    
    # 汇总
    puts "\n========================================="
    puts "  Errors:   $errors"
    puts "  Warnings: $warnings"
    puts "========================================="
    
    return [expr {$errors == 0}]
}
```

```python
#!/usr/bin/env python3
"""
Power Distribution Visualizer
使用文本方式展示功耗分布（无需matplotlib依赖）
"""

from typing import Dict, List, Tuple


def text_bar_chart(data: List[Tuple[str, float]], 
                   title: str = "Power Distribution",
                   width: int = 50):
    """生成文本柱状图"""
    if not data:
        return
    
    max_val = max(v for _, v in data)
    total = sum(v for _, v in data)
    
    print(f"\n{'='*60}")
    print(f"  {title}")
    print(f"  Total: {total:.2f} mW")
    print(f"{'='*60}")
    
    for name, value in data:
        bar_len = int(value / max_val * width) if max_val > 0 else 0
        pct = value / total * 100 if total > 0 else 0
        bar = '█' * bar_len + '░' * (width - bar_len)
        print(f"  {name:20s} │{bar}│ {value:8.2f} mW ({pct:5.1f}%)")
    
    print(f"{'='*60}")


def power_breakdown_chart(report):
    """生成功耗组成分析图"""
    total_internal = sum(m.internal_power for m in report.modules)
    total_switching = sum(m.switching_power for m in report.modules)
    total_leakage = sum(m.leakage_power for m in report.modules)
    
    data = [
        ("Internal Power", total_internal),
        ("Switching Power", total_switching),
        ("Leakage Power", total_leakage),
    ]
    
    text_bar_chart(data, "Power Type Breakdown")


def module_ranking_chart(report, top_n: int = 10):
    """生成模块功耗排名图"""
    sorted_modules = sorted(report.modules,
                           key=lambda m: m.total_power,
                           reverse=True)
    
    data = [(m.hierarchy, m.total_power) 
            for m in sorted_modules[:top_n]]
    
    text_bar_chart(data, f"Top {top_n} Power Consumers")
```

```python
#!/usr/bin/env python3
"""
Power Analysis CI/CD Integration
集成到持续集成流程中的功耗检查脚本
"""

import sys
import argparse
from pathlib import Path


def power_ci_check(report_path: str, 
                    budget_mw: float,
                    cg_target: float = 90.0,
                    lvt_limit: float = 10.0) -> bool:
    """CI/CD功耗检查入口"""
    
    all_passed = True
    
    print("=" * 60)
    print("  Power Analysis CI Check")
    print("=" * 60)
    
    # 1. 总功耗预算检查
    report = parse_ptpx_report(report_path)
    print(f"\n[1] Total Power Budget Check")
    print(f"    Budget: {budget_mw:.2f} mW")
    print(f"    Actual: {report.total_power:.2f} mW")
    
    if report.total_power <= budget_mw:
        print(f"    ✅ PASS")
    else:
        print(f"    ❌ FAIL (over budget by "
              f"{report.total_power - budget_mw:.2f} mW)")
        all_passed = False
    
    # 2. 功耗回归检查
    print(f"\n[2] Power Regression Check")
    tracker = PowerTrendTracker()
    versions = tracker.data.get('versions', [])
    if versions:
        prev = versions[-1]['total_power']
        delta_pct = (report.total_power - prev) / prev * 100
        print(f"    Previous: {prev:.2f} mW")
        print(f"    Current:  {report.total_power:.2f} mW")
        print(f"    Delta:    {delta_pct:+.1f}%")
        
        if delta_pct > 5.0:
            print(f"    ⚠️  WARNING: >5% regression")
        else:
            print(f"    ✅ PASS")
    
    # 3. 结果
    print(f"\n{'='*60}")
    if all_passed:
        print("  🎉 All power checks PASSED")
    else:
        print("  ❌ Some power checks FAILED")
    print(f"{'='*60}")
    
    return all_passed


if __name__ == '__main__':
    parser = argparse.ArgumentParser(
        description='Power Analysis CI Check')
    parser.add_argument('report', help='PTPX report file')
    parser.add_argument('--budget', type=float, required=True,
                       help='Power budget in mW')
    parser.add_argument('--cg-target', type=float, default=90.0,
                       help='Clock gating coverage target %%')
    parser.add_argument('--lvt-limit', type=float, default=10.0,
                       help='Maximum LVT cell percentage')
    
    args = parser.parse_args()
    
    passed = power_ci_check(
        args.report, args.budget,
        args.cg_target, args.lvt_limit
    )
    
    sys.exit(0 if passed else 1)
```

```makefile
power-trend: $(POWER_REPORT)
	python3 scripts/power_trend.py \
```
