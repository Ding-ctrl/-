```python
import subprocess
import json

def extract_rtl_features(rtl_file, top_module):
    """
    从RTL文件提取功耗相关特征
    使用Yosys进行分析
    """
    # Yosys合成脚本
    yosys_script = f"""
    read_verilog {rtl_file}
    hierarchy -top {top_module}
    proc; opt; fsm; opt; memory; opt
    techmap; opt
    stat
    """
    
    # 执行Yosys并解析输出
    result = subprocess.run(
        ['yosys', '-p', yosys_script],
        capture_output=True, text=True
    )
    
    # 解析统计数据（示例解析逻辑）
    features = {}
    for line in result.stdout.split('\n'):
        if 'Number of wires:' in line:
            features['num_wires'] = int(line.split(':')[1].strip())
        elif 'Number of cells:' in line:
            features['num_cells'] = int(line.split(':')[1].strip())
        elif '$dff' in line:
            features['num_ff'] = int(line.split(':')[1].strip())
        elif '$mul' in line:
            features['num_mul'] = int(line.split(':')[1].strip())
        elif '$add' in line:
            features['num_add'] = int(line.split(':')[1].strip())
    
    # 计算派生特征
    features['ff_ratio'] = features.get('num_ff', 0) / max(features.get('num_cells', 1), 1)
    features['arithmetic_ratio'] = (
        features.get('num_mul', 0) + features.get('num_add', 0)
    ) / max(features.get('num_cells', 1), 1)
    
    return features
```

```python
import torch
import torch.nn as nn
from torch_geometric.nn import GCNConv, global_mean_pool

class PowerGNN(nn.Module):
    """
    图神经网络功耗预测模型
    将RTL网表建模为图，节点=门/触发器，边=连线
    """
    def __init__(self, node_features=16, hidden_dim=64):
        super().__init__()
        
        # 图卷积层（聚合邻居信息）
        self.conv1 = GCNConv(node_features, hidden_dim)
        self.conv2 = GCNConv(hidden_dim, hidden_dim)
        self.conv3 = GCNConv(hidden_dim, hidden_dim // 2)
        
        # 全局池化后的预测层
        self.predictor = nn.Sequential(
            nn.Linear(hidden_dim // 2, 32),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(32, 1)  # 输出：功耗预测值（W）
        )
    
    def forward(self, x, edge_index, batch):
        # 图卷积（提取拓扑特征）
        x = torch.relu(self.conv1(x, edge_index))
        x = torch.relu(self.conv2(x, edge_index))
        x = torch.relu(self.conv3(x, edge_index))
        
        # 全局池化（将图聚合为向量）
        x = global_mean_pool(x, batch)
        
        # 功耗预测
        power = self.predictor(x)
        return power.squeeze()
```

### 32.2.2 功耗预测模型的训练与评估

```python
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_absolute_percentage_error

# 简单而有效的方法：基于提取特征的梯度提升树
# （对小数据集比GNN更鲁棒）

class PowerPredictionPipeline:
    def __init__(self):
        self.scaler = StandardScaler()
        self.model = GradientBoostingRegressor(
            n_estimators=200,
            learning_rate=0.05,
            max_depth=5,
            subsample=0.8,
            random_state=42
        )
    
    def train(self, X_features, y_power):
        """
        X_features: shape (n_designs, n_features)
        y_power: shape (n_designs,) - 真实功耗（W），来自PrimeTime PX
        """
        X_train, X_val, y_train, y_val = train_test_split(
            X_features, y_power, test_size=0.2, random_state=42
        )
        
        X_train_scaled = self.scaler.fit_transform(X_train)
        X_val_scaled = self.scaler.transform(X_val)
        
        self.model.fit(X_train_scaled, y_train)
        
        # 评估
        y_pred = self.model.predict(X_val_scaled)
        mape = mean_absolute_percentage_error(y_val, y_pred) * 100
        
        print(f"验证集 MAPE: {mape:.1f}%")
        print(f"功耗范围: {y_val.min():.2f}W ~ {y_val.max():.2f}W")
        
        return mape
    
    def predict(self, X_new):
        X_scaled = self.scaler.transform(X_new)
        return self.model.predict(X_scaled)
    
    def feature_importance(self, feature_names):
        importance = self.model.feature_importances_
        sorted_idx = np.argsort(importance)[::-1]
        
        print("\n功耗预测中最重要的特征：")
        for i in range(min(10, len(feature_names))):
            idx = sorted_idx[i]
            print(f"  {i+1}. {feature_names[idx]}: {importance[idx]:.3f}")
```

### 32.3.1 DVFS参数的RL优化

**问题建模：**

```python
import gym
import numpy as np
from gym import spaces

class DVFSOptimizationEnv(gym.Env):
    """
    将DVFS优化建模为强化学习环境
    目标：在性能约束下最小化能耗
    """
    
    def __init__(self, workload_profiles):
        super().__init__()
        
        self.workloads = workload_profiles
        
        # 动作空间：选择VF工作点
        # (freq_level, voltage_level) 各5个选项
        self.action_space = spaces.Discrete(25)  # 5×5 VF点
        
        # 状态空间：当前负载特征
        self.observation_space = spaces.Box(
            low=0, high=1,
            shape=(10,),  # 10个特征
            dtype=np.float32
        )
        
        # 可用的VF工作点
        self.vf_points = [
            (0.4, 0.6), (0.6, 0.65), (0.8, 0.7),
            (1.0, 0.75), (1.2, 0.8), (1.5, 0.85),
            (1.8, 0.9), (2.0, 0.95), (2.5, 1.0), (3.0, 1.05)
            # (freq_GHz, voltage_V)
        ]
        
        self.current_step = 0
        self.max_steps = 1000
    
    def _get_state(self, workload):
        """将负载特征转换为状态向量"""
        return np.array([
            workload['cpu_util'],
            workload['mem_bandwidth'],
            workload['cache_miss_rate'],
            workload['ipc'],
            workload['branch_miss_rate'],
            workload['fp_util'],
            workload['temporal_locality'],
            workload['spatial_locality'],
            workload['thread_count'] / 8.0,
            workload['priority'],
        ], dtype=np.float32)
    
    def step(self, action):
        freq, volt = self.vf_points[action]
        workload = self.workloads[self.current_step]
        
        # 计算性能
        effective_perf = min(1.0, workload['cpu_util'] * (freq / 3.0))
        
        # 计算功耗（简化模型）
        alpha = workload['cpu_util'] * 0.1
        C = 50e-15
        power = alpha * C * volt**2 * (freq * 1e9) + 0.05 * volt**4
        
        # 奖励：满足性能需求前提下最小化功耗
        perf_requirement = workload.get('perf_req', 0.7)
        if effective_perf >= perf_requirement:
            reward = -power  # 满足性能，奖励与功耗成反比
        else:
            reward = -100 * (perf_requirement - effective_perf) - power
        
        self.current_step += 1
        done = self.current_step >= min(len(self.workloads), self.max_steps)
        
        next_workload = self.workloads[min(self.current_step, len(self.workloads)-1)]
        state = self._get_state(next_workload)
        
        return state, reward, done, {'power': power, 'perf': effective_perf}
    
    def reset(self):
        self.current_step = 0
        return self._get_state(self.workloads[0])
```

### 32.5.1 操作数隔离的自动识别

**自动识别需要添加操作数隔离的信号：**

```python
import re
from sklearn.tree import DecisionTreeClassifier

def auto_detect_iso_candidates(design_data):
    """
    自动识别适合添加操作数隔离的信号
    
    设计数据来源：
    - 功耗仿真：每个节点的切换活动
    - 功能仿真：使能信号的有效率
    - 综合结果：每个操作单元的驱动/负载关系
    """
    candidates = []
    
    for cell_name, cell_data in design_data['cells'].items():
        # 提取特征
        features = {
            'toggle_rate': cell_data['toggle_rate'],
            'enable_coverage': cell_data.get('enable_signal_coverage', 0),
            'is_arithmetic': cell_data['type'] in ['MUL', 'ADD', 'DIV', 'FP_ALU'],
            'input_fanin': cell_data['fanin'],
            'output_fanout': cell_data['fanout'],
            'power_contribution': cell_data['power_mW'],
            'enable_toggle_rate': cell_data.get('enable_toggle_rate', 0.5),
        }
        
        # 规则引导的启发式方法（也可训练分类器）
        score = 0
        
        # 高功耗算术单元 + 低使能覆盖 → 高优先级
        if features['is_arithmetic'] and features['power_contribution'] > 1.0:
            score += 30
        
        # 使能信号覆盖率低（<50%时，操作数隔离效果好）
        if features['enable_coverage'] < 0.5:
            score += 20 * (1 - features['enable_coverage'])
        
        # 扇出大：隔离后节省更多
        if features['output_fanout'] > 10:
            score += 10
        
        if score > 20:
            candidates.append({
                'cell': cell_name,
                'score': score,
                'estimated_savings_mW': (
                    features['power_contribution'] * 
                    (1 - features['enable_toggle_rate'])
                ),
                'enable_coverage': features['enable_coverage'],
            })
    
    # 按节省功耗排序
    candidates.sort(key=lambda x: x['estimated_savings_mW'], reverse=True)
    
    print("操作数隔离候选（按功耗节省排序）：")
    print(f"{'模块':<30} {'优先级分数':<12} {'预估节省':<15} {'使能覆盖率'}")
    for c in candidates[:20]:  # 显示前20个
        print(f"{c['cell']:<30} {c['score']:<12} "
              f"{c['estimated_savings_mW']:.1f}mW       "
              f"{c['enable_coverage']:.1%}")
    
    return candidates
```

### 32.5.2 ML驱动的功耗优化闭环

```python
from bayes_opt import BayesianOptimization

def power_aware_synthesis_optimization():
    """
    贝叶斯优化综合参数，最小化功耗同时满足时序
    """
    
    def run_synthesis_and_measure(
        clock_gating_effort,  # 0.0-1.0
        optimization_effort,  # 0.0-1.0 
        max_fanout,           # 8-32
        vt_mix_ratio,         # 0.0-1.0 (HVT ratio)
    ):
        """调用综合工具，返回功耗（越高越差，取负值用于最大化）"""
        
        # 将连续参数转换为工具可接受的设置
        cg_level = 'high' if clock_gating_effort > 0.7 else (
                   'medium' if clock_gating_effort > 0.3 else 'low')
        
        opt_level = int(optimization_effort * 3) + 1  # 1-4
        fanout = int(max_fanout * 24 + 8)  # 8-32
        
        # 运行综合（调用Design Compiler/Genus）
        power_mW, timing_slack_ps = call_synthesis_tool(
            cg_level=cg_level,
            opt_level=opt_level,
            max_fanout=fanout,
            hvt_ratio=vt_mix_ratio
        )
        
        # 时序违规的惩罚
        timing_penalty = max(0, -timing_slack_ps * 10) if timing_slack_ps < 50 else 0
        
        # 返回负功耗（贝叶斯优化最大化目标）
        return -(power_mW + timing_penalty)
    
    # 参数空间
    pbounds = {
        'clock_gating_effort': (0.0, 1.0),
        'optimization_effort': (0.0, 1.0),
        'max_fanout': (0.0, 1.0),
        'vt_mix_ratio': (0.0, 1.0),
    }
    
    optimizer = BayesianOptimization(
        f=run_synthesis_and_measure,
        pbounds=pbounds,
        random_state=42,
    )
    
    # 初始随机探索
    optimizer.maximize(init_points=5, n_iter=0)
    
    # 贝叶斯优化迭代（每次迭代≈一次综合运行）
    optimizer.maximize(init_points=0, n_iter=20)
    
    print(f"\n最优参数：{optimizer.max['params']}")
    print(f"最优功耗：{-optimizer.max['target']:.1f} mW")
    
    return optimizer.max
```

## 32.6 量化感知训练（QAT）与低功耗推理

### 32.6.1 量化对推理功耗的影响

```python
import torch
import torch.nn as nn

class PowerAwareQuantization:
    """
    量化感知训练与功耗估算
    """
    
    @staticmethod
    def estimate_power_by_precision(model, input_shape, mac_power_per_op):
        """
        估算不同精度下的推理功耗
        
        mac_power_per_op: 每次MAC操作的能耗（pJ）
        各精度典型值（7nm工艺）：
          FP32: 3.7 pJ/MAC
          FP16: 1.9 pJ/MAC  
          INT8: 0.5 pJ/MAC
          INT4: 0.14 pJ/MAC
          INT2: 0.04 pJ/MAC (理论值，实际对精度损失大)
        """
        power_table = {
            'fp32': 3.7,
            'fp16': 1.9, 
            'bf16': 2.0,
            'int8': 0.5,
            'int4': 0.14,
        }
        
        # 计算模型的MAC数（乘加运算次数）
        total_macs = PowerAwareQuantization._count_macs(model, input_shape)
        
        print(f"\n模型总MAC数：{total_macs/1e9:.2f} GMACs")
        print(f"\n各精度下的推理能耗（每次推理）：")
        print(f"{'精度':<8} {'每MAC能耗(pJ)':<15} {'总能耗(μJ)':<15} {'相对FP32'}")
        print("-" * 55)
        
        fp32_energy = total_macs * power_table['fp32']
        
        for prec, pj_per_mac in power_table.items():
            total_energy_uj = total_macs * pj_per_mac / 1e6  # 转μJ
            ratio = pj_per_mac / power_table['fp32']
            print(f"{prec:<8} {pj_per_mac:<15.1f} {total_energy_uj:<15.1f} {ratio:.2f}×")
    
    @staticmethod
    def _count_macs(model, input_shape):
        """简单的MAC计数（实际可用ptflops库）"""
        total = 0
        x = torch.zeros(1, *input_shape)
        
        for module in model.modules():
            if isinstance(module, nn.Conv2d):
                # Conv2d MACs = Cout × Cin × Kh × Kw × H_out × W_out
                out_h = (input_shape[-2] + 2*module.padding[0] - module.kernel_size[0]) // module.stride[0] + 1
                out_w = (input_shape[-1] + 2*module.padding[1] - module.kernel_size[1]) // module.stride[1] + 1
                macs = module.out_channels * module.in_channels * \
                       module.kernel_size[0] * module.kernel_size[1] * out_h * out_w
                total += macs
            elif isinstance(module, nn.Linear):
                total += module.in_features * module.out_features
        
        return total
```
